  </Card>
  <Card title="Εργαλεία για κάθε ιδέα" icon="open-book">

  </Card>

<CardGrid>
  <Card title="Τοπικά από προεπιλογή" icon="laptop">

  </Card>
  <Card title="Χρησιμοποιήστε το δικό σας cloud" icon="rocket">

  </Card>
